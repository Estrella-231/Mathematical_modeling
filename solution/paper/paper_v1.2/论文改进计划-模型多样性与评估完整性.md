```python
from scipy import stats
import numpy as np

# 假设我们有三种规则下的FFI序列
ffi_rank_sum = np.array([...])  # 241个周的FFI值
ffi_percent_sum = np.array([...])
ffi_judge_save = np.array([...])

# 1. 配对t检验（Rank Sum vs Percent Sum）
t_stat, p_value = stats.ttest_rel(ffi_rank_sum, ffi_percent_sum)
print(f"Paired t-test: t={t_stat:.4f}, p={p_value:.4f}")

# 2. Wilcoxon符号秩检验（非参数版本）
w_stat, w_pvalue = stats.wilcoxon(ffi_rank_sum, ffi_percent_sum)
print(f"Wilcoxon test: W={w_stat}, p={w_pvalue:.4f}")

# 3. Cohen's d效应量
def cohens_d(x, y):
    nx, ny = len(x), len(y)
    dof = nx + ny - 2
    return (np.mean(x) - np.mean(y)) / np.sqrt(((nx-1)*np.std(x, ddof=1)**2 + 
                                                 (ny-1)*np.std(y, ddof=1)**2) / dof)

d_rank_percent = cohens_d(ffi_rank_sum, ffi_percent_sum)
print(f"Cohen's d: {d_rank_percent:.4f}")

# 4. 置信区间（Bootstrap）
from scipy.stats import bootstrap

def mean_diff(x, y):
    return np.mean(x) - np.mean(y)

rng = np.random.default_rng(42)
res = bootstrap((ffi_rank_sum, ffi_percent_sum), 
                lambda x, y: mean_diff(x, y),
                n_resamples=10000, 
                confidence_level=0.95,
                random_state=rng)

print(f"95% CI for mean difference: [{res.confidence_interval.low:.4f}, "
      f"{res.confidence_interval.high:.4f}]")
```

```python
from sklearn.utils import resample
import numpy as np

def bootstrap_prediction_interval(model, X_train, y_train, X_test, 
                                   n_iterations=1000, confidence=0.95):
    """
    使用Bootstrap方法构建预测区间
    """
    predictions = []
    
    for i in range(n_iterations):
        # 重采样训练集
        X_resampled, y_resampled = resample(X_train, y_train, random_state=i)
        
        # 重新训练模型
        model_boot = clone(model)
        model_boot.fit(X_resampled, y_resampled)
        
        # 预测
        y_pred = model_boot.predict(X_test)
        predictions.append(y_pred)
    
    predictions = np.array(predictions)
    
    # 计算置信区间
    alpha = (1 - confidence) / 2
    lower = np.percentile(predictions, alpha * 100, axis=0)
    upper = np.percentile(predictions, (1 - alpha) * 100, axis=0)
    median = np.percentile(predictions, 50, axis=0)
    
    return median, lower, upper

# 应用到Q1预测
from sklearn.base import clone

median_pred, lower_bound, upper_bound = bootstrap_prediction_interval(
    ridge_model, X_train, y_train, X_test, 
    n_iterations=1000, confidence=0.95
)

# 计算区间宽度统计
interval_widths = upper_bound - lower_bound
print(f"Average interval width: {np.mean(interval_widths):.4f}")
print(f"Median interval width: {np.median(interval_widths):.4f}")
print(f"Coverage rate (test set): {np.mean((y_test >= lower_bound) & (y_test <= upper_bound)):.2%}")
```

```python
import matplotlib.pyplot as plt
from scipy import stats
import numpy as np

def plot_residual_diagnostics(y_true, y_pred, title_prefix=""):
    """
    生成4张残差诊断图
    """
    residuals = y_true - y_pred
    
    fig, axes = plt.subplots(2, 2, figsize=(14, 12))
    
    # 1. 残差vs拟合值（检验同方差性）
    axes[0, 0].scatter(y_pred, residuals, alpha=0.6)
    axes[0, 0].axhline(y=0, color='r', linestyle='--')
    axes[0, 0].set_xlabel('Fitted Values')
    axes[0, 0].set_ylabel('Residuals')
    axes[0, 0].set_title(f'{title_prefix}Residuals vs. Fitted')
    axes[0, 0].grid(True, alpha=0.3)
    
    # 添加局部回归平滑线（检测非线性）
    from scipy.interpolate import UnivariateSpline
    sorted_idx = np.argsort(y_pred)
    spline = UnivariateSpline(y_pred[sorted_idx], residuals[sorted_idx], s=100)
    axes[0, 0].plot(y_pred[sorted_idx], spline(y_pred[sorted_idx]), 
                    'g-', linewidth=2, label='Smoothed trend')
    axes[0, 0].legend()
    
    # 2. Q-Q图（检验正态性）
    stats.probplot(residuals, dist="norm", plot=axes[0, 1])
    axes[0, 1].set_title(f'{title_prefix}Normal Q-Q Plot')
    axes[0, 1].grid(True, alpha=0.3)
    
    # 3. Scale-Location图（检验同方差性）
    standardized_residuals = residuals / np.std(residuals)
    axes[1, 0].scatter(y_pred, np.sqrt(np.abs(standardized_residuals)), alpha=0.6)
    axes[1, 0].set_xlabel('Fitted Values')
    axes[1, 0].set_ylabel('√|Standardized Residuals|')
    axes[1, 0].set_title(f'{title_prefix}Scale-Location')
    axes[1, 0].grid(True, alpha=0.3)
    
    # 4. 残差直方图 + 核密度估计（检验正态性）
    axes[1, 1].hist(residuals, bins=30, density=True, alpha=0.6, color='blue')
    
    # 拟合正态分布曲线
    mu, sigma = np.mean(residuals), np.std(residuals)
    x = np.linspace(residuals.min(), residuals.max(), 100)
    axes[1, 1].plot(x, stats.norm.pdf(x, mu, sigma), 'r-', linewidth=2, 
                    label=f'Normal(μ={mu:.2f}, σ={sigma:.2f})')
    
    # KDE曲线
    from scipy.stats import gaussian_kde
    kde = gaussian_kde(residuals)
    axes[1, 1].plot(x, kde(x), 'g--', linewidth=2, label='KDE')
    
    axes[1, 1].set_xlabel('Residuals')
    axes[1, 1].set_ylabel('Density')
    axes[1, 1].set_title(f'{title_prefix}Residual Distribution')
    axes[1, 1].legend()
    axes[1, 1].grid(True, alpha=0.3)
    
    plt.tight_layout()
    return fig

# 应用到Ridge模型
y_pred_ridge = ridge_model.predict(X_test)
fig = plot_residual_diagnostics(y_test, y_pred_ridge, title_prefix="Ridge - ")
plt.savefig('residual_diagnostics_ridge.png', dpi=300)

# 正态性检验（Shapiro-Wilk）
residuals = y_test - y_pred_ridge
shapiro_stat, shapiro_p = stats.shapiro(residuals)
```

```python
from sklearn.model_selection import cross_validate, KFold
import numpy as np

def comprehensive_cross_validation(model, X, y, cv=5):
    """
    全面的交叉验证，返回多个指标
    """
    scoring = {
        'r2': 'r2',
        'neg_mse': 'neg_mean_squared_error',
        'neg_mae': 'neg_mean_absolute_error',
        'neg_rmse': 'neg_root_mean_squared_error'
    }
    
    cv_results = cross_validate(model, X, y, cv=cv, scoring=scoring,
                                 return_train_score=True, n_jobs=-1)
    
    # 整理结果
    results_summary = {
        'train_r2_mean': cv_results['train_r2'].mean(),
        'train_r2_std': cv_results['train_r2'].std(),
        'test_r2_mean': cv_results['test_r2'].mean(),
        'test_r2_std': cv_results['test_r2'].std(),
        'test_rmse_mean': np.sqrt(-cv_results['test_neg_mse'].mean()),
        'test_rmse_std': np.sqrt(-cv_results['test_neg_mse']).std(),
        'test_mae_mean': -cv_results['test_neg_mae'].mean(),
        'test_mae_std': (-cv_results['test_neg_mae']).std(),
```

```python
from sklearn.model_selection import TimeSeriesSplit

def time_series_cv_by_season(model, X, y, seasons, n_splits=5):
    """
    按赛季顺序的时间序列交叉验证
    """
    tscv = TimeSeriesSplit(n_splits=n_splits)
    
    results = []
    for fold, (train_idx, test_idx) in enumerate(tscv.split(X), 1):
        X_train_fold, X_test_fold = X[train_idx], X[test_idx]
        y_train_fold, y_test_fold = y[train_idx], y[test_idx]
        
        # 训练
        model_fold = clone(model)
        model_fold.fit(X_train_fold, y_train_fold)
        
        # 评估
        y_pred_fold = model_fold.predict(X_test_fold)
        r2_fold = r2_score(y_test_fold, y_pred_fold)
        rmse_fold = np.sqrt(mean_squared_error(y_test_fold, y_pred_fold))
        
        results.append({
            'Fold': fold,
            'Train_Seasons': f"S{seasons[train_idx[0]]}-S{seasons[train_idx[-1]]}",
            'Test_Seasons': f"S{seasons[test_idx[0]]}-S{seasons[test_idx[-1]]}",
            'Train_Size': len(train_idx),
            'Test_Size': len(test_idx),
            'R²': r2_fold,
            'RMSE': rmse_fold
        })
    
    return pd.DataFrame(results)

# 应用
tscv_results = time_series_cv_by_season(ridge_model, X, y, seasons_array, n_splits=5)
```

```python
from sklearn.model_selection import learning_curve

def plot_learning_curve(estimator, X, y, title, cv=5):
    train_sizes, train_scores, test_scores = learning_curve(
        estimator, X, y, cv=cv,
        train_sizes=np.linspace(0.1, 1.0, 10),
        scoring='r2',
        n_jobs=-1
    )
    
    train_mean = np.mean(train_scores, axis=1)
    train_std = np.std(train_scores, axis=1)
    test_mean = np.mean(test_scores, axis=1)
    test_std = np.std(test_scores, axis=1)
    
    plt.figure(figsize=(10, 6))
    plt.plot(train_sizes, train_mean, 'o-', color='r', label='Training score')
    plt.fill_between(train_sizes, train_mean - train_std, train_mean + train_std,
                     alpha=0.1, color='r')
    
    plt.plot(train_sizes, test_mean, 'o-', color='g', label='Cross-validation score')
    plt.fill_between(train_sizes, test_mean - test_std, test_mean + test_std,
                     alpha=0.1, color='g')
    
    plt.xlabel('Training Set Size')
    plt.ylabel('R² Score')
    plt.title(f'Learning Curve: {title}')
    plt.legend(loc='best')
    plt.grid(True, alpha=0.3)
    
    # 添加收敛判断
    final_gap = train_mean[-1] - test_mean[-1]
    if final_gap < 0.1:
        convergence_text = "Well converged (gap < 0.1)"
    elif final_gap < 0.2:
        convergence_text = "Moderate convergence (gap < 0.2)"
    else:
        convergence_text = "Poor convergence (gap ≥ 0.2)"
    
    plt.text(0.5, 0.05, convergence_text, 
             transform=plt.gca().transAxes,
             bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.5))
    
    return plt.gcf()

# 应用到主要模型

plt.savefig('learning_curve_ridge.png', dpi=300)
```

```python
# evaluation/metrics_summary.py

import numpy as np
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error

def compute_all_metrics(y_true, y_pred, model_name="Model"):
    """
    计算所有评估指标
    """
    metrics = {
        'Model': model_name,
        'R²': r2_score(y_true, y_pred),
        'RMSE': np.sqrt(mean_squared_error(y_true, y_pred)),
        'MAE': mean_absolute_error(y_true, y_pred),
        'MAPE': np.mean(np.abs((y_true - y_pred) / y_true)) * 100,
        'Max Error': np.max(np.abs(y_true - y_pred)),
        'Explained Variance': explained_variance_score(y_true, y_pred),
    }
    
    return metrics

def compare_models(models_dict, X_test, y_test):
    """
    对比多个模型
    
    Args:
        models_dict: {'Ridge': ridge_model, 'Lasso': lasso_model, ...}
        X_test: 测试集特征
        y_test: 测试集标签
    
    Returns:
        pd.DataFrame: 对比表格
    """
    results = []
    
    for name, model in models_dict.items():
        y_pred = model.predict(X_test)
        metrics = compute_all_metrics(y_test, y_pred, model_name=name)
        results.append(metrics)
    
    df = pd.DataFrame(results)
    
    # 高亮最优值
    for col in ['R²', 'Explained Variance']:
        df[col] = df[col].apply(lambda x: f"\\textbf{{{x:.4f}}}" if x == df[col].max() else f"{x:.4f}")
    
    for col in ['RMSE', 'MAE', 'MAPE', 'Max Error']:
        df[col] = df[col].apply(lambda x: f"\\textbf{{{x:.4f}}}" if x == df[col].min() else f"{x:.4f}")
    
    return df
```
